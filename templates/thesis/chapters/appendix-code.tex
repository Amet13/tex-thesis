\unnumberedsection{Appendix A. Sample code listing}\label{app:code}

The notification module used by the monitoring system (file \texttt{notifier.py}) is reproduced below.
Long code in an appendix is set without a float, so it may run over several pages.

\begin{verbatim}
"""Notification module for the monitoring system."""

import json
import logging
import smtplib
import urllib.request
from email.mime.text import MIMEText

logger = logging.getLogger(__name__)


class EmailNotifier:
    """Send notifications via email."""

    def __init__(self, host: str, port: int, sender: str):
        self.host, self.port, self.sender = host, port, sender

    def send(self, subject: str, body: str,
             to: list[str]) -> bool:
        msg = MIMEText(body)
        msg["Subject"], msg["From"] = subject, self.sender
        msg["To"] = ", ".join(to)
        try:
            with smtplib.SMTP(self.host, self.port) as server:
                server.send_message(msg)
            return True
        except OSError as err:
            logger.error("Email failed: %s", err)
            return False


class WebhookNotifier:
    """Send notifications via a JSON webhook."""

    def __init__(self, url: str):
        self.url = url

    def send(self, subject: str, body: str) -> bool:
        data = json.dumps({"title": subject, "text": body})
        req = urllib.request.Request(
            self.url, data=data.encode(),
            headers={"Content-Type": "application/json"},
        )
        try:
            urllib.request.urlopen(req, timeout=10)
            return True
        except OSError as err:
            logger.error("Webhook failed: %s", err)
            return False
\end{verbatim}
