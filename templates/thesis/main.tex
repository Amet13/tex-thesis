% The first line turns on PDF tagging and declares PDF/UA-2 (accessible PDF)

%   lang          document language (screen readers, hyphenation metadata)
%   pdfstandard   ua-2 = PDF/UA-2 (ISO 14289-2), checked by veraPDF
%   pdfversion    PDF/UA-2 requires PDF 2.0
%   tagging       structure tags for headings, lists, tables, figures, links
%   math/setup    mathml-SE: formulas become MathML structure (LuaLaTeX)

  pdfstandard = ua-2,
  pdfversion  = 2.0,

  tagging-setup = {math/setup=mathml-SE},
