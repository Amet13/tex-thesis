% Research paper template — Overfull (https://github.com/amet13/overfull)
%
% Builds with LuaLaTeX (default: `latexmk`) and with pdfLaTeX
% (`latexmk -pdf`, needed for arXiv, which does not run LuaLaTeX).
%
% \DocumentMetadata turns on PDF tagging and declares PDF/UA-2 conformance:
%   lang         document language
%   pdfstandard  ua-2 = PDF/UA-2 (accessible PDF), checked by veraPDF
%   pdfversion   PDF/UA-2 requires PDF 2.0
%   tagging      structure tags for headings, lists, tables, figures, links
%   math/setup   mathml-SE: formulas as MathML structure (LuaLaTeX only;
%                with pdfLaTeX the formula's LaTeX source becomes its alt text)
\ifdefined\directlua
  \DocumentMetadata{lang=en, pdfstandard=ua-2, pdfversion=2.0, tagging=on,
    tagging-setup={math/setup=mathml-SE}}
\else
  \DocumentMetadata{lang=en, pdfstandard=ua-2, pdfversion=2.0, tagging=on}
\fi
% Paper size: a4paper or letterpaper. Body size: 10pt, 11pt or 12pt.
\documentclass[11pt,a4paper]{article}

\input{metadata}
\input{preamble}

\title{\PaperTitle}
\author{\PaperAuthors\\[2pt]
  \small\PaperAffiliations\\
  \small\texttt{\PaperEmail}}
\date{} % no date on submissions

\begin{document}
\maketitle

\begin{abstract}
  State the problem, what you did, the main result with its key number, and why it matters, in 150--250 words.
  Write it last.
\end{abstract}

\noindent\textbf{Keywords:} \PaperKeywords

\input{sections/introduction}
\input{sections/related}
\input{sections/method}
\input{sections/results}
\input{sections/conclusion}

% End of the main text. Page limits in overfull.yml (venue.pageLimit) are
% counted up to this label; references and appendices usually do not count.
\label{overfull:main-end}

\input{sections/ai-disclosure}

\printbibliography

\appendix
\input{sections/appendix}

\end{document}
